\sectitle{第二部分：产品介绍}

\setstretch{1.3}
\textbf{股票图表分析工具}是一款面向普通投资者的辅助分析软件，主要提供以下功能：

\begin{enumerate}[leftmargin=2em]
  \item \textbf{K线图查看}：展示股票的历史价格走势
  \item \textbf{技术指标分析}：MACD、RSI、布林带、均线等
  \item \textbf{股票筛选}：按照价格、涨跌幅、成交量等条件筛选
  \item \textbf{AI预测功能}（实验性）：基于历史数据的涨跌方向预测
\end{enumerate}

\vspace{1em}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!50!black,title=\textbf{⚙️ 技术说明},fonttitle=\bfseries\large]
\setstretch{1.2}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{数据来源}：新浪财经、腾讯财经公开数据
  \item \textbf{数据存储}：本地SQLite数据库，保护您的隐私
  \item \textbf{AI模型}：基于规则引擎 + 机器学习（随机森林）
  \item \textbf{运行环境}：Windows 10/11，需安装 Edge WebView2 运行时
\end{itemize}
\end{tcolorbox}

\sectitle{第三部分：AI预测功能说明（实验性功能）}

\begin{tcolorbox}[colback=yellow!10!white,colframe=yellow!70!black,title=\textbf{⚠️ AI预测特别声明},fonttitle=\bfseries\large]
\setstretch{1.2}
\textbf{\large 重要提醒：AI预测结果仅供学习参考，不代表投资建议！}
\begin{itemize}[leftmargin=1.5em]
  \item 本工具的AI功能基于\textbf{历史数据规律}，但股票市场受政策、突发事件、情绪等多种因素影响
  \item \textbf{模型准确率有限}，历史回测不代表未来表现
  \item \textbf{切勿仅凭AI预测进行投资决策}！请结合基本面、消息面综合判断
  \item 建议将AI预测作为\textbf{众多参考指标之一}，而非唯一依据
\end{itemize}
\end{tcolorbox}

\vspace{1em}
\setstretch{1.3}
\textbf{AI预测功能采用以下技术：}

\begin{itemize}[leftmargin=1.5em]
  \item \textbf{特征工程}：计算50+维技术指标特征
  \item \textbf{信号系统}：均线、MACD、RSI、布林带、动量等多维度信号
  \item \textbf{评分机制}：综合多个信号给出-10到+10的评分
  \item \textbf{预测结果}：强烈看多、看多、中性偏多、中性偏空、看空、强烈看空
\end{itemize}

\vspace{1em}

\begin{tcolorbox}[colback=lightbg,colframe=darkwarning,title=\textbf{📊 预测结果解读},fonttitle=\bfseries\large]
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{得分 > 4}：强烈看多 —— 多个技术指标共振看涨
  \item \textbf{得分 2-4}：看多 —— 偏乐观信号较多
  \item \textbf{得分 -2 ~ 2}：中性 —— 信号不明确，建议观望
  \item \textbf{得分 -4 ~ -2}：看空 —— 偏悲观信号较多
  \item \textbf{得分 < -4}：强烈看空 —— 多个技术指标共振看跌
\end{itemize}

\vspace{0.5em}
\textbf{请注意}：以上评分标准基于历史数据统计，\textbf{不代表未来走势}。
\end{tcolorbox}

\sectitle{第四部分：使用指南（大字版）}

\setstretch{1.4}
\large

\textbf{一、启动软件}
\begin{enumerate}[leftmargin=2em]
  \item 双击桌面图标，等待3-5秒启动
  \item 首次使用请先点击\textbf{「开始预加载」}按钮
  \item 预加载约需5-10分钟（取决于网络速度）
\end{enumerate}

\vspace{1em}
\textbf{二、查看股票K线}
\begin{enumerate}[leftmargin=2em]
  \item 在左侧输入股票代码（如：000001）
  \item 点击\textbf{「查看股票」}按钮
  \item 图表区域将显示K线走势
  \item 可切换日K/5分钟分时
\end{enumerate}

\vspace{1em}
\textbf{三、使用AI预测}
\begin{enumerate}[leftmargin=2em]
  \item 点击右上角\textbf{「AI预测」}按钮
  \item 输入股票代码，点击\textbf{「AI预测」}
  \item 查看预测结果和各项信号
\end{enumerate}

\vspace{1em}
\textbf{四、筛选潜力股}
\begin{enumerate}[leftmargin=2em]
  \item 在AI预测模式下，点击\textbf{「开始筛选」}
  \item 等待筛选完成
  \item 点击任意股票可直接查看K线图
\end{enumerate}

\vspace{1em}
\textbf{五、调整显示}
\begin{enumerate}[leftmargin=2em]
  \item \textbf{字号调整}：右上角小/中/大/特大按钮
  \item \textbf{列表高度}：拖拽股票列表下方手柄
  \item \textbf{导出图表}：点击图表区「导出图表」按钮
\end{enumerate}
